\newlength{\llmkLogoUnit}
\newcommand{\llmkLogo}[1][1]{\bgroup
  % scale
  \pgfmathsetlength{\llmkLogoUnit}{0.2cm * (#1)}%
  % colors (Arcade Bloom palette)
  % - Primary: red (RGB: 191, 1, 8; HEX: BF0108)
  \definecolor{color1}{RGB}{191, 1, 8}%
  % - Accent 2: blue (RGB: 1, 113, 160; HEX: 0171A0)
  \definecolor{color2}{RGB}{1, 113, 160}%
  % - Accent 1: yellow (RGB: 245, 154, 2; HEX: F59A02)
  \definecolor{color3}{RGB}{245, 154, 2}%
  % - Main text: black (RGB: 66, 66, 66; HEX: 424242)
  \definecolor{colorLine}{RGB}{66, 66, 66}%
  % shapes
  \pgfdeclarelayer{bg}%
  \pgfsetlayers{bg,main}%
  \tikzstyle{octagon} = [
    shape = regular polygon,
    regular polygon sides = 8
  ]%
  \tikzstyle{star} = [
    shape = star,
    star point ratio = 1.3,
    star points = 8
  ]%
  \begin{tikzpicture}[scale=#1, every node/.style={scale=#1}]
  \node [circle, minimum width = {5\llmkLogoUnit}, fill = color1] (n1) {};
  \node [octagon, minimum width = {5\llmkLogoUnit}, fill = color2, right = {2\llmkLogoUnit of n1}] (n2) {};
  \node [star, minimum width = {5\llmkLogoUnit}, fill = color3, right = {2\llmkLogoUnit of n2}] (n3) {};
  \begin{pgfonlayer}{bg}
  \draw [colorLine, line width=.25\llmkLogoUnit, shorten <= -\llmkLogoUnit, shorten >= -\llmkLogoUnit] (n1) -- (n3);
  \end{pgfonlayer}%
  \end{tikzpicture}\egroup}

% EOF
